# =============================================================================
# MYPOS website: environment variables (template)
#
# Copy this file to:
#   .env.local  read by the website (next dev / next build / next start)
#   .env        read by the Prisma CLI and the scripts in prisma/
# Both files are gitignored. NEVER put real values in this file: it is committed.
#
# Format rules:
#   - One KEY=value per line, no spaces around "=".
#   - A value that contains # or spaces must be in double quotes, otherwise
#     everything after # is dropped as a comment.
#     Example: ADMIN_PASSWORD="abc#123"
#   - Leave a value empty (KEY=) to turn an optional feature off.
# =============================================================================


# --- Database (required) ------------------------------------------------------

# PostgreSQL connection string (Supabase: Project Settings > Database >
# Connection string, "Transaction pooler"). Contains the DB password.

# --- Admin login (required) ---------------------------------------------------

# Signs the admin login cookie. Any long random text, at least 32 characters.
# Generate: openssl rand -base64 48
# Changing it signs every admin out.
SESSION_SECRET=

# Optional: true / false. Leave empty (recommended): the login cookie is
# HTTPS-only when the site is opened over HTTPS, and also works over plain
# http on a test server (http://<IP>:3000).
ADMIN_COOKIE_SECURE=

# Optional: needed only to turn on two-step login (authenticator app).
# base64 of 32 random bytes. Generate: openssl rand -base64 32
# Do not reuse SESSION_SECRET. If this key is lost or changed, every account
# with two-step login must be reset with prisma/reset-2fa.ts.
TWO_FACTOR_ENCRYPTION_KEY=

# Used only once by prisma/seed-admin-user.ts to create the first admin
# account. After that, manage admins in /admin/admin-users and you may
# remove these two values.
ADMIN_EMAIL=
ADMIN_PASSWORD=


# --- Uploads: photos and PDFs (Supabase Storage, required for the admin) ------

SUPABASE_URL=

# Server-only secret key (Supabase: Project Settings > API Keys, secret / service_role).
# Full access to the project: never expose it in the browser or in chat.
SUPABASE_SERVICE_ROLE_KEY=

# Storage bucket for uploads. Default: uploads
SUPABASE_STORAGE_BUCKET=uploads
